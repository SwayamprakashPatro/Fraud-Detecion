SELECT
  t.Transaction_ID,
  t.Amount,
  t.Channel,
  t.Transaction_Timestamp,
  t.Sender_Account_ID   AS Sender_Account,
  t.Receiver_Account_ID AS Receiver_Account,
  t.Device_ID,
  t.Merchant_ID,
  t.UPI_ID,
  f.Fraud_Type,
  COALESCE(f.Label, 0)  AS Label
FROM `TRANSACTION` t
LEFT JOIN FRAUD_CASE f
  ON t.Transaction_ID = f.Transaction_ID;
    # Run from the project root, next to train.py. Needs the repo's requirements.
import pickle, numpy as np, pandas as pd, torch
from src.config import config
from src.data_loader import FraudGraphDataset
from src.models.graphsage import FraudGraphSAGE

ds = FraudGraphDataset()
data, meta = ds.load_data()
model = FraudGraphSAGE()
ck = torch.load(config.BEST_MODEL_PATH, map_location="cpu")
model.load_state_dict(ck["model_state_dict"])
model.eval()
prob = model.predict_proba(data.x, data.edge_index, data.tx_src,
                           data.tx_dst, data.tx_edge_attr).numpy()

# Same ordering as the loader: Cleaned_Database.csv sorted by Transaction_ID.
ids = (pd.read_csv(config.CSV_DATA_PATH, usecols=["Transaction_ID"])
         ["Transaction_ID"].sort_values().astype(int).tolist())
split = np.where(data.test_mask.numpy(), "test",
        np.where(data.val_mask.numpy(), "val", "train"))
info = {}
G = pickle.load(open(config.GRAPH_PATH, "rb"))
for u, v, k, d in G.edges(keys=True, data=True):
    row = info.setdefault(int(k), {})
    t = d.get("type")
    if t == "transaction":
        row.update(Sender_Account=u.split("_")[1], Receiver_Account=v.split("_")[1],
                   Amount=d.get("amount"), Channel=d.get("channel"),
                   Transaction_Timestamp=d.get("timestamp"),
                   Fraud_Type=d.get("fraud_type") if d.get("label") == 1 else None)
    elif t == "used_device": row["Device_ID"] = v.split("_")[1]
    elif t == "merchant":    row["Merchant_ID"] = v.split("_")[1]
    elif t == "upi_used":    row["UPI_ID"] = v.split("_")[1]
rows = []
for i, tid in enumerate(ids[:len(prob)]):
    rows.append({"Transaction_ID": tid, **info.get(tid, {}),
                 "Predicted_Prob": round(float(prob[i]), 4), "Split": split[i]})
pd.DataFrame(rows).to_csv("fraud_data.csv", index=False)

    Shapes are kinds of dot

    Circle: account. Diamond: shop. Square: device. Triangle: UPI ID. Turn on Show devices, shops and UPI to see the last three. Bigger means more payments.

    Lines are payments

    Thicker means more money. Coloured lines are fraud, one colour per kind. Grey lines are normal payments.

    Crowds are a clue

    Many fraud lines meeting at one dot can mean stolen money is collected there. Dots with a red ring are the biggest hotspots. Hover any dot for the full story.
